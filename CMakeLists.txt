cmake_minimum_required(VERSION 3.16)
project(RobotDrawingLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "This application uses Win32 and Windows OpenGL.")
endif()
add_executable(RobotDrawingLab WIN32 src/main.cpp src/svg_import.cpp)
target_link_libraries(RobotDrawingLab PRIVATE opengl32 glu32 gdiplus comctl32 comdlg32 gdi32 ole32 uuid)
if(MINGW)
  target_link_options(RobotDrawingLab PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
if(MSVC)
  target_compile_options(RobotDrawingLab PRIVATE /utf-8 /W4)
endif()
